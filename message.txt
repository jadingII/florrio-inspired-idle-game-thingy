"""
Particle Accelerator & Physics Simulator
=========================================
Simulates a synchrotron-style circular accelerator with realistic physics:

  - Relativistic kinematics  (Lorentz factor γ, relativistic momentum)
  - Lorentz force            (F = q(E + v×B))
  - Dipole bending magnets   (keeps beam on circular path)
  - Quadrupole focusing      (prevents beam divergence)
  - RF acceleration cavities (boost energy each revolution)
  - Synchrotron radiation    (energy loss per turn)
  - Beam-beam collisions     (detect & visualise interaction point)
  - Multiple particle species: proton, electron, positron, muon, lead ion

Controls
--------
  Space       pause / resume
  R           reset
  +/-         speed up / slow down simulation
  A           toggle RF acceleration
  B           toggle dipole magnets
  Q           toggle quadrupole focusing
  C           trigger manual collision
  S           toggle synchrotron radiation loss
  Tab         cycle particle species
  1-5         set number of particles (1/5/20/100/500)
  G           toggle grid / field visualisation
  F           toggle force vectors
  Scroll      zoom
  Drag        pan
  Escape      quit
"""

import sys, math, random, time
import numpy as np
import pygame

# ── Constants (SI) ────────────────────────────────────────────────────────────
C        = 2.997924e8      # speed of light  m/s
E_CHARGE = 1.602176e-19    # elementary charge  C
M_PROTON = 1.672622e-27    # kg
M_ELEC   = 9.109384e-31    # kg
M_MUON   = 1.883532e-28    # kg
M_LEAD   = 207.2 * 1.660539e-27   # Pb-208 nucleus kg  (charge 82e)
EV       = E_CHARGE        # 1 eV in joules
GEV      = 1e9 * EV

# ── Particle species definitions ──────────────────────────────────────────────
SPECIES = {
    'proton':   dict(mass=M_PROTON,    charge= E_CHARGE, color=(100,180,255), name='Proton',   symbol='p⁺'),
    'electron': dict(mass=M_ELEC,      charge=-E_CHARGE, color=(255,120, 60), name='Electron', symbol='e⁻'),
    'positron': dict(mass=M_ELEC,      charge= E_CHARGE, color=( 60,255,120), name='Positron', symbol='e⁺'),
    'muon':     dict(mass=M_MUON,      charge=-E_CHARGE, color=(220,100,255), name='Muon',     symbol='μ⁻'),
    'lead':     dict(mass=M_LEAD,      charge=82*E_CHARGE,color=(255,200, 60),name='Pb-208',   symbol='Pb⁸²⁺'),
}
SPECIES_KEYS = list(SPECIES.keys())

# ── Window ────────────────────────────────────────────────────────────────────
W, H   = 1280, 780
FPS    = 60
TITLE  = 'Particle Accelerator Simulator'

# ── Accelerator geometry (in metres, rendered scaled) ────────────────────────
RING_R     = 4.0      # ring radius  m
N_DIPOLES  = 8        # bending magnets
N_QUADS    = 16       # quadrupole lenses
N_RF       = 2        # RF cavities
RF_VOLTAGE = 5e6      # V  per cavity per turn
B_FIELD    = 1.2      # T  dipole field
QUAD_GRAD  = 15.0     # T/m  quadrupole gradient


# ── Physics helpers ───────────────────────────────────────────────────────────

def gamma(v, c=C):
    beta2 = min((v/c)**2, 1-1e-12)
    return 1.0 / math.sqrt(1 - beta2)

def lorentz_gamma_from_KE(KE_joules, mass):
    return 1.0 + KE_joules / (mass * C**2)

def speed_from_gamma(g):
    return C * math.sqrt(1 - 1/g**2)

def relativistic_radius(mass, charge, KE_joules, B):
    """Cyclotron radius for relativistic particle."""
    g   = lorentz_gamma_from_KE(KE_joules, mass)
    p   = g * mass * speed_from_gamma(g)
    return p / (abs(charge) * B) if B > 0 else RING_R

def synchrotron_loss(gamma_val, mass, charge, radius):
    """Energy radiated per revolution (GeV), classical formula."""
    # U₀ = C_γ * E⁴ / (ρ * (m₀c²)⁴)   simplified scalar form
    E_total  = gamma_val * mass * C**2
    Cgamma   = 8.85e-5  # m/GeV³ for electrons; scaled by (m_e/m)³
    mass_fac = (M_ELEC / mass)**3
    U_joules = Cgamma * mass_fac * (E_total/GEV)**4 / radius * GEV
    return U_joules


# ── Particle ──────────────────────────────────────────────────────────────────

class Particle:
    _id = 0

    def __init__(self, species_key, KE_joules, angle_offset=0.0, radial_offset=0.0):
        Particle._id += 1
        self.pid      = Particle._id
        self.species  = species_key
        sp            = SPECIES[species_key]
        self.mass     = sp['mass']
        self.charge   = sp['charge']
        self.color    = sp['color']

        self.KE       = KE_joules           # kinetic energy joules
        self.angle    = angle_offset        # position angle on ring (rad)
        self.r        = RING_R + radial_offset  # current radius

        self._update_speed()

        # Velocity in 2-D world coords
        vx = -self.speed * math.sin(self.angle)
        vy =  self.speed * math.cos(self.angle)
        self.vel      = np.array([vx, vy], dtype=float)

        # Position
        self.pos      = np.array([math.cos(self.angle)*self.r,
                                   math.sin(self.angle)*self.r], dtype=float)

        self.alive    = True
        self.trail    = []
        self.rev      = 0          # revolutions completed
        self.last_ang = self.angle
        self.energy_history = [KE_joules / GEV]

    def _update_speed(self):
        g          = lorentz_gamma_from_KE(self.KE, self.mass)
        self.gamma = g
        self.speed = speed_from_gamma(g)
        self.beta  = self.speed / C

    def kinetic_energy_GeV(self):
        return self.KE / GEV

    def total_energy_GeV(self):
        return (self.gamma * self.mass * C**2) / GEV

    def momentum_GeV_c(self):
        return (self.gamma * self.mass * self.speed) / (GEV / C)


# ── Simulation ────────────────────────────────────────────────────────────────

class Accelerator:
    def __init__(self):
        self.species_idx  = 0
        self.species_key  = SPECIES_KEYS[self.species_idx]

        self.KE0          = 6.5e12 * EV   # 6.5 TeV LHC-like starting energy
        self.particles: list[Particle] = []
        self.n_target     = 1

        self.rf_on        = True
        self.dipole_on    = True
        self.quad_on      = True
        self.rad_loss_on  = True
        self.show_grid    = True
        self.show_forces  = False

        self.paused       = False
        self.sim_speed    = 1.0
        self.dt_base      = 1e-9       # simulation timestep seconds
        self.t            = 0.0

        self.collisions: list[dict] = []   # recent collision events
        self.col_flash              = 0.0

        self.spawn_particles(1)

        # Dipole positions (angles)
        self.dipole_angles = [2*math.pi*i/N_DIPOLES for i in range(N_DIPOLES)]
        self.quad_angles   = [2*math.pi*i/N_QUADS   for i in range(N_QUADS)]
        self.rf_angles     = [2*math.pi*i/N_RF       for i in range(N_RF)]

    def spawn_particles(self, n):
        self.particles.clear()
        Particle._id = 0
        for i in range(n):
            offset_a = random.uniform(-0.05, 0.05)
            offset_r = random.uniform(-0.05, 0.05)
            ke       = self.KE0 * random.uniform(0.98, 1.0)
            self.particles.append(Particle(self.species_key, ke, offset_a, offset_r))
        self.n_target = n

    def reset(self):
        self.t = 0.0
        self.collisions.clear()
        self.col_flash = 0.0
        self.spawn_particles(self.n_target)

    def step(self):
        if self.paused:
            return
        dt = self.dt_base * self.sim_speed
        self.t += dt

        for p in self.particles:
            if not p.alive:
                continue

            # ── Dipole bending (centripetal Lorentz force) ─────────────────
            if self.dipole_on:
                r_hat  = p.pos / (np.linalg.norm(p.pos) + 1e-30)
                # Magnetic force magnitude: F = qvB (centripetal)
                Fmag   = abs(p.charge) * p.speed * B_FIELD
                # Direction: inward (−r̂) for positive charge moving CCW
                sign   = 1.0 if p.charge > 0 else -1.0
                F_dip  = -sign * Fmag * r_hat
            else:
                F_dip  = np.zeros(2)

            # ── Quadrupole focusing ────────────────────────────────────────
            # Restoring force toward ring centre proportional to radial displacement
            if self.quad_on:
                r_now     = np.linalg.norm(p.pos)
                dr        = r_now - RING_R
                r_hat2    = p.pos / (r_now + 1e-30)
                Fq_mag    = abs(p.charge) * p.speed * QUAD_GRAD * abs(dr)
                F_quad    = -np.sign(dr) * Fq_mag * r_hat2
            else:
                F_quad    = np.zeros(2)

            # ── RF acceleration ────────────────────────────────────────────
            # Each time particle crosses an RF cavity angle, give energy kick
            if self.rf_on:
                for ang in self.rf_angles:
                    p_ang = math.atan2(p.pos[1], p.pos[0]) % (2*math.pi)
                    diff  = abs(p_ang - ang) % (2*math.pi)
                    diff  = min(diff, 2*math.pi - diff)
                    if diff < 0.05:
                        dE = abs(p.charge) * RF_VOLTAGE / N_RF * self.sim_speed * dt * 1e8
                        p.KE += dE

            # ── Synchrotron radiation loss ─────────────────────────────────
            if self.rad_loss_on and p.gamma > 1.001:
                loss_per_s = synchrotron_loss(p.gamma, p.mass, p.charge, RING_R) / (2*math.pi*RING_R/p.speed)
                p.KE       = max(p.KE * 0.5, p.KE - loss_per_s * dt)

            # ── Update kinematics ──────────────────────────────────────────
            p._update_speed()

            # Total force
            F_total = F_dip + F_quad

            # Relativistic acceleration: a = F / (γm)
            a = F_total / (p.gamma * p.mass + 1e-60)

            # Update velocity, re-normalise to current speed
            p.vel += a * dt
            spd    = np.linalg.norm(p.vel)
            if spd > 1e-10:
                p.vel = p.vel / spd * p.speed

            # Update position
            p.pos += p.vel * dt

            # Track revolution count
            new_ang = math.atan2(p.pos[1], p.pos[0])
            old_ang = math.atan2(p.vel[1]*(-dt) + p.pos[1] - p.vel[1]*dt,
                                  p.vel[0]*(-dt) + p.pos[0] - p.vel[0]*dt)
            # Simple revolution counter via sign change of x near y>0
            if p.pos[1] > 0 and (p.pos - p.vel*dt)[1] <= 0:
                p.rev += 1
                p.energy_history.append(p.kinetic_energy_GeV())
                if len(p.energy_history) > 300:
                    p.energy_history.pop(0)

            # Trail
            p.trail.append(tuple(p.pos))
            if len(p.trail) > 120:
                p.trail.pop(0)

        # ── Collision detection (opposite-sign charge particles) ───────────
        pos_list = [(p, p.pos) for p in self.particles if p.alive and p.charge > 0]
        neg_list = [(p, p.pos) for p in self.particles if p.alive and p.charge < 0]
        for pp, ppos in pos_list:
            for np_, npos in neg_list:
                dist = np.linalg.norm(ppos - npos)
                if dist < 0.15:
                    ecm = (pp.total_energy_GeV() + np_.total_energy_GeV())
                    self.collisions.append(dict(
                        pos=tuple((ppos+npos)/2),
                        ecm=ecm,
                        t=self.t,
                    ))
                    self.col_flash = 1.0
                    if len(self.collisions) > 20:
                        self.collisions.pop(0)

        self.col_flash = max(0.0, self.col_flash - 0.02 * self.sim_speed)


# ── Renderer ──────────────────────────────────────────────────────────────────

class Renderer:
    def __init__(self, accel: Accelerator):
        pygame.init()
        self.screen = pygame.display.set_mode((W, H), pygame.RESIZABLE)
        pygame.display.set_caption(TITLE)
        self.clock  = pygame.time.Clock()
        self.accel  = accel

        # Camera
        self.cam_x  = W / 2
        self.cam_y  = H / 2 + 30
        self.scale  = 70.0    # pixels per metre

        self.dragging = False
        self.drag_last = (0, 0)

        self.font_sm  = pygame.font.SysFont('monospace', 12)
        self.font_med = pygame.font.SysFont('monospace', 14, bold=True)
        self.font_lg  = pygame.font.SysFont('monospace', 20, bold=True)
        self.font_xl  = pygame.font.SysFont('monospace', 28, bold=True)

    # ── Coord helpers ──────────────────────────────────────────────────────

    def w2s(self, wx, wy):
        """World (metres) → screen pixels."""
        return (int(self.cam_x + wx * self.scale),
                int(self.cam_y - wy * self.scale))

    def s2w(self, sx, sy):
        return ((sx - self.cam_x) / self.scale,
                (self.cam_y - sy) / self.scale)

    # ── Drawing ────────────────────────────────────────────────────────────

    def draw_background(self):
        W2, H2 = self.screen.get_size()
        self.screen.fill((6, 6, 18))

        if self.accel.show_grid:
            spacing = self.scale
            if spacing < 8:
                spacing *= 5
            ox = self.cam_x % spacing
            oy = self.cam_y % spacing
            col = (20, 20, 50)
            for x in range(int(ox), W2, int(spacing)):
                pygame.draw.line(self.screen, col, (x, 0), (x, H2))
            for y in range(int(oy), H2, int(spacing)):
                pygame.draw.line(self.screen, col, (0, y), (W2, y))

    def draw_field_map(self):
        """Draw B-field strength as background heatmap (sampled grid)."""
        if not self.accel.show_grid:
            return
        W2, H2 = self.screen.get_size()
        step = 40
        for sx in range(0, W2, step):
            for sy in range(0, H2, step):
                wx, wy = self.s2w(sx, sy)
                r  = math.sqrt(wx**2 + wy**2)
                dr = abs(r - RING_R)
                # Field strongest near ring
                strength = max(0, 1 - dr / 0.8)
                if strength > 0.05:
                    col = (int(0*strength), int(40*strength), int(120*strength))
                    s = pygame.Surface((step, step), pygame.SRCALPHA)
                    s.fill(col + (int(60*strength),))
                    self.screen.blit(s, (sx, sy))

    def draw_ring(self):
        cx, cy = self.w2s(0, 0)
        r_px   = int(RING_R * self.scale)
        # Beam pipe
        pygame.draw.circle(self.screen, (30, 60, 100), (cx, cy), r_px, max(2, int(self.scale*0.18)))
        pygame.draw.circle(self.screen, (50, 90, 160), (cx, cy), r_px, 2)

    def draw_magnets(self):
        a    = self.accel
        r_px = RING_R * self.scale

        for ang in a.dipole_angles:
            mx = math.cos(ang) * RING_R
            my = math.sin(ang) * RING_R
            sx, sy = self.w2s(mx, my)
            size   = max(6, int(self.scale * 0.25))
            col    = (60, 180, 255) if a.dipole_on else (60, 80, 100)
            pygame.draw.rect(self.screen, col, (sx-size, sy-size, size*2, size*2))
            pygame.draw.rect(self.screen, (120, 220, 255) if a.dipole_on else (80,100,120),
                             (sx-size, sy-size, size*2, size*2), 2)
            lbl = self.font_sm.render('D', True, (200,240,255))
            self.screen.blit(lbl, (sx-4, sy-6))

        for ang in a.quad_angles:
            mx = math.cos(ang) * RING_R
            my = math.sin(ang) * RING_R
            sx, sy = self.w2s(mx, my)
            size   = max(4, int(self.scale * 0.15))
            col    = (255, 180, 60) if a.quad_on else (80, 60, 30)
            pts    = [(sx, sy-size), (sx+size, sy), (sx, sy+size), (sx-size, sy)]
            pygame.draw.polygon(self.screen, col, pts)
            pygame.draw.polygon(self.screen, (255,220,140) if a.quad_on else (100,80,40), pts, 2)

        for ang in a.rf_angles:
            mx = math.cos(ang) * RING_R
            my = math.sin(ang) * RING_R
            sx, sy = self.w2s(mx, my)
            size   = max(5, int(self.scale * 0.2))
            col    = (60, 255, 120) if a.rf_on else (30, 80, 50)
            pygame.draw.ellipse(self.screen, col, (sx-size, sy-size//2, size*2, size))
            pygame.draw.ellipse(self.screen, (140,255,180) if a.rf_on else (60,100,70),
                                (sx-size, sy-size//2, size*2, size), 2)
            lbl = self.font_sm.render('RF', True, (200,255,220))
            self.screen.blit(lbl, (sx-8, sy-6))

    def draw_particles(self):
        a = self.accel
        for p in a.particles:
            if not p.alive:
                continue

            # Trail
            if len(p.trail) > 2:
                trail_pts = [self.w2s(tx, ty) for tx, ty in p.trail]
                for i in range(1, len(trail_pts)):
                    alpha = i / len(trail_pts)
                    col   = tuple(int(c * alpha * 0.7) for c in p.color)
                    pygame.draw.line(self.screen, col, trail_pts[i-1], trail_pts[i], 1)

            sx, sy = self.w2s(p.pos[0], p.pos[1])
            size   = max(3, int(self.scale * 0.1))

            # Glow
            glow = pygame.Surface((size*6, size*6), pygame.SRCALPHA)
            gc   = tuple(min(255, int(c*0.6)) for c in p.color)
            pygame.draw.circle(glow, gc + (40,), (size*3, size*3), size*3)
            self.screen.blit(glow, (sx-size*3, sy-size*3))

            pygame.draw.circle(self.screen, p.color, (sx, sy), size)
            pygame.draw.circle(self.screen, (255,255,255), (sx, sy), max(1, size//2))

            # Force vector
            if a.show_forces and len(p.trail) > 1:
                old = p.trail[-2]
                dv  = p.pos - np.array(old)
                scale_f = self.scale * 3e8
                ex  = sx + int(dv[1] * scale_f)
                ey  = sy - int(dv[0] * scale_f)
                pygame.draw.line(self.screen, (255,80,80), (sx,sy), (ex,ey), 2)

    def draw_collisions(self):
        a   = self.accel
        now = a.t
        for col in a.collisions:
            age = now - col['t']
            if age > 2e-8:
                continue
            sx, sy   = self.w2s(*col['pos'])
            fade     = max(0, 1 - age / 2e-8)
            r        = int(self.scale * 0.8 * fade)
            if r < 2:
                continue
            burst = pygame.Surface((r*4, r*4), pygame.SRCALPHA)
            for ring_r, ring_a in [(r, 180), (r//2, 120), (r//4, 80)]:
                pygame.draw.circle(burst, (255, 220, 80, int(ring_a*fade)),
                                   (r*2, r*2), ring_r, 2)
            self.screen.blit(burst, (sx-r*2, sy-r*2))

            lbl = self.font_sm.render(f'ECM={col["ecm"]:.1f} GeV', True, (255,240,120))
            self.screen.blit(lbl, (sx+r+4, sy-8))

        # Flash overlay
        if a.col_flash > 0.01:
            fl = pygame.Surface(self.screen.get_size(), pygame.SRCALPHA)
            fl.fill((255, 220, 60, int(a.col_flash * 40)))
            self.screen.blit(fl, (0, 0))

    def draw_hud(self):
        a     = self.accel
        W2, H2 = self.screen.get_size()
        sp    = SPECIES[a.species_key]

        # ── Left panel ──────────────────────────────────────────────────────
        lines = [
            ('ACCELERATOR', (160, 200, 255), self.font_lg),
            (f'Species:   {sp["name"]} ({sp["symbol"]})', sp['color'], self.font_med),
            ('', None, self.font_sm),
        ]
        if a.particles:
            p = a.particles[0]
            lines += [
                (f'KE:        {p.kinetic_energy_GeV():.3f} GeV',    (200,255,200), self.font_med),
                (f'E_total:   {p.total_energy_GeV():.3f} GeV',      (200,255,200), self.font_med),
                (f'p:         {p.momentum_GeV_c():.3f} GeV/c',      (200,230,255), self.font_med),
                (f'γ (gamma): {p.gamma:.4g}',                        (255,220,160), self.font_med),
                (f'β (v/c):   {p.beta:.8f}',                         (255,220,160), self.font_med),
                (f'Speed:     {p.speed/C*100:.6f}% c',               (255,200,120), self.font_med),
                (f'Revs:      {p.rev}',                               (180,180,255), self.font_med),
                ('', None, self.font_sm),
            ]
        lines += [
            (f'Particles: {len(a.particles)}',      (200,200,255), self.font_med),
            (f'Sim time:  {a.t:.3e} s',             (180,180,200), self.font_med),
            (f'Sim speed: {a.sim_speed:.1f}x',      (180,180,200), self.font_med),
            (f'Collisions:{len(a.collisions)}',     (255,200,80),  self.font_med),
        ]

        panel = pygame.Surface((260, len(lines)*20 + 16), pygame.SRCALPHA)
        panel.fill((8, 8, 24, 200))
        self.screen.blit(panel, (8, 8))

        y = 14
        for text, col, font in lines:
            if text:
                surf = font.render(text, True, col or (200,200,200))
                self.screen.blit(surf, (14, y))
            y += 20

        # ── Right panel: controls ─────────────────────────────────────────
        def st(flag): return ('ON ', (80,255,120)) if flag else ('OFF', (255,80,80))

        rf_s,  rf_c  = st(a.rf_on)
        dip_s, dip_c = st(a.dipole_on)
        qd_s,  qd_c  = st(a.quad_on)
        rad_s, rad_c = st(a.rad_loss_on)
        gr_s,  gr_c  = st(a.show_grid)
        fv_s,  fv_c  = st(a.show_forces)
        pau_s, pau_c = ('PAUSED', (255,200,80)) if a.paused else ('RUNNING',(80,255,120))

        ctrl_lines = [
            (f'[Space] {pau_s}',          pau_c),
            (f'[A] RF cavities:   {rf_s}',  rf_c),
            (f'[B] Dipole magnets:{dip_s}', dip_c),
            (f'[Q] Quadrupoles:  {qd_s}',   qd_c),
            (f'[S] Synch. loss:  {rad_s}',  rad_c),
            (f'[G] Grid/field:   {gr_s}',   gr_c),
            (f'[F] Force vecs:   {fv_s}',   fv_c),
            ('', None),
            ('[Tab] Cycle species', (200,200,255)),
            ('[1-5] 1/5/20/100/500 ptcl', (200,200,255)),
            ('[+/-] Sim speed',     (200,200,255)),
            ('[C]   Manual collision',(255,200,80)),
            ('[R]   Reset',          (255,120,120)),
            ('[Scroll/Drag] View',   (160,160,200)),
        ]

        cx0 = W2 - 258
        panel2 = pygame.Surface((252, len(ctrl_lines)*18 + 16), pygame.SRCALPHA)
        panel2.fill((8, 8, 24, 200))
        self.screen.blit(panel2, (cx0 - 4, 8))

        y = 14
        for text, col in ctrl_lines:
            if text:
                surf = self.font_sm.render(text, True, col or (180,180,200))
                self.screen.blit(surf, (cx0, y))
            y += 18

        # ── Energy history plot ───────────────────────────────────────────
        if a.particles and len(a.particles[0].energy_history) > 2:
            hist = a.particles[0].energy_history
            ph   = 80
            pw   = 220
            px0  = 14
            py0  = H2 - ph - 40

            bg = pygame.Surface((pw, ph), pygame.SRCALPHA)
            bg.fill((8,8,24,200))
            self.screen.blit(bg, (px0, py0))

            pygame.draw.rect(self.screen, (40,40,80), (px0, py0, pw, ph), 1)
            lbl = self.font_sm.render('Energy/rev (GeV)', True, (160,160,200))
            self.screen.blit(lbl, (px0+2, py0+2))

            emin = min(hist); emax = max(hist)
            rng  = max(emax-emin, 1e-6)
            pts  = []
            for i, e in enumerate(hist):
                hx = px0 + int(i / max(len(hist)-1,1) * (pw-4)) + 2
                hy = py0 + ph - int((e-emin)/rng*(ph-16)) - 8
                pts.append((hx, hy))
            if len(pts) > 1:
                sp_col = SPECIES[a.species_key]['color']
                pygame.draw.lines(self.screen, sp_col, False, pts, 2)

            ev_lbl = self.font_sm.render(f'{hist[-1]:.2f} GeV', True, SPECIES[a.species_key]['color'])
            self.screen.blit(ev_lbl, (px0+pw-70, py0+ph-16))

        # ── Status bar ────────────────────────────────────────────────────
        bar = pygame.Surface((W2, 26), pygame.SRCALPHA)
        bar.fill((8,8,24,200))
        self.screen.blit(bar, (0, H2-26))

                  f'Quad grad={QUAD_GRAD}T/m  |  '
                  f'FPS={int(self.clock.get_fps())}')
        surf = self.font_sm.render(status, True, (120,120,180))
        self.screen.blit(surf, (8, H2-20))

    def draw(self):
        self.draw_background()
        self.draw_field_map()
        self.draw_ring()
        self.draw_magnets()
        self.draw_particles()
        self.draw_collisions()
        self.draw_hud()
        pygame.display.flip()

    # ── Events ────────────────────────────────────────────────────────────

    def handle_events(self):
        a = self.accel
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                pygame.quit(); sys.exit()

            elif event.type == pygame.VIDEORESIZE:
                self.screen = pygame.display.set_mode((event.w, event.h), pygame.RESIZABLE)

            elif event.type == pygame.KEYDOWN:
                k = event.key
                if   k == pygame.K_SPACE:  a.paused = not a.paused
                elif k == pygame.K_r:      a.reset()
                elif k == pygame.K_a:      a.rf_on       = not a.rf_on
                elif k == pygame.K_b:      a.dipole_on   = not a.dipole_on
                elif k == pygame.K_q:      a.quad_on     = not a.quad_on
                elif k == pygame.K_s:      a.rad_loss_on = not a.rad_loss_on
                elif k == pygame.K_g:      a.show_grid   = not a.show_grid
                elif k == pygame.K_f:      a.show_forces = not a.show_forces
                elif k == pygame.K_TAB:
                    a.species_idx = (a.species_idx + 1) % len(SPECIES_KEYS)
                    a.species_key = SPECIES_KEYS[a.species_idx]
                    a.reset()
                elif k == pygame.K_c:
                    # Force collision by placing two opposite-charge particles close
                    if len(a.particles) >= 2:
                        a.particles[1].pos = a.particles[0].pos.copy() + np.array([0.1, 0.0])
                elif k == pygame.K_EQUALS or k == pygame.K_PLUS:
                    a.sim_speed = min(200.0, a.sim_speed * 1.5)
                elif k == pygame.K_MINUS:
                    a.sim_speed = max(0.1, a.sim_speed / 1.5)
                elif k == pygame.K_1: a.n_target=1;   a.spawn_particles(1)
                elif k == pygame.K_2: a.n_target=5;   a.spawn_particles(5)
                elif k == pygame.K_3: a.n_target=20;  a.spawn_particles(20)
                elif k == pygame.K_4: a.n_target=100; a.spawn_particles(100)
                elif k == pygame.K_5: a.n_target=500; a.spawn_particles(500)
                elif k == pygame.K_ESCAPE:
                    pygame.quit(); sys.exit()

            elif event.type == pygame.MOUSEBUTTONDOWN:
                if event.button == 2:
                    self.dragging = True
                    self.drag_last = event.pos
                elif event.button == 4:
                    self.scale = min(300.0, self.scale * 1.15)
                elif event.button == 5:
                    self.scale = max(10.0,  self.scale / 1.15)

            elif event.type == pygame.MOUSEBUTTONUP:
                if event.button == 2:
                    self.dragging = False

            elif event.type == pygame.MOUSEMOTION:
                if self.dragging:
                    dx = event.pos[0] - self.drag_last[0]
                    dy = event.pos[1] - self.drag_last[1]
                    self.cam_x += dx
                    self.cam_y += dy
                    self.drag_last = event.pos

    def run(self):
        STEPS_PER_FRAME = 20
        while True:
            self.handle_events()
            for _ in range(STEPS_PER_FRAME):
                self.accel.step()
            self.draw()
            self.clock.tick(FPS)
